add_executable(s4_backend_test_runner backend_test_main.cpp)
target_include_directories(s4_backend_test_runner PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(s4_backend_test_runner PRIVATE S4)
if(NOT TARGET infrastructure::mekil AND NOT TARGET mekil)
    message(FATAL_ERROR "s4_backend_test_runner requires infrastructure::mekil (or mekil)")
endif()
if(TARGET infrastructure::mekil)
    target_link_libraries(s4_backend_test_runner PRIVATE infrastructure::mekil)
else()
    target_link_libraries(s4_backend_test_runner PRIVATE mekil)
endif()
if(NOT MSVC AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(s4_backend_test_runner PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
    target_link_options(s4_backend_test_runner PRIVATE -fsanitize=address)
endif()
